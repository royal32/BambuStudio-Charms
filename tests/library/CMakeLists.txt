cmake_minimum_required(VERSION 3.16)
project(SlicedLibraryChecks LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
foreach(check IN ITEMS storage task)
    add_executable(library-${check}-checks ../sliced_library_${check}.test.cpp)
    target_compile_features(library-${check}-checks PRIVATE cxx_std_17)
    target_include_directories(library-${check}-checks PRIVATE ../../src)
    target_link_libraries(library-${check}-checks PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(library-${check}-checks PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(library-${check}-checks PRIVATE -UNDEBUG)
    endif()
    add_test(NAME library-${check} COMMAND library-${check}-checks)
    set_tests_properties(library-${check} PROPERTIES TIMEOUT 15)
endforeach()
