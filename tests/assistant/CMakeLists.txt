cmake_minimum_required(VERSION 3.16)
project(StudioAssistantChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
find_package(OpenSSL REQUIRED)
find_package(Boost REQUIRED)
if(APPLE)
    # Test networking uses the system curl. The application keeps its bundled curl.
    find_library(ASSISTANT_CURL curl REQUIRED PATHS "${CMAKE_OSX_SYSROOT}/usr/lib" NO_DEFAULT_PATH)
else()
    find_package(CURL REQUIRED)
    set(ASSISTANT_CURL CURL::libcurl)
endif()
add_executable(studio-assistant-tests ../studio_assistant.test.cpp
    "${ROOT}/src/slic3r/GUI/StudioAssistantClient.cpp"
    "${ROOT}/src/slic3r/GUI/StudioAssistantTools.cpp")
target_include_directories(studio-assistant-tests PRIVATE "${ROOT}/src" ${Boost_INCLUDE_DIRS})
target_link_libraries(studio-assistant-tests PRIVATE OpenSSL::Crypto ${ASSISTANT_CURL})
target_compile_definitions(studio-assistant-tests PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
if(WIN32)
    target_link_libraries(studio-assistant-tests PRIVATE ws2_32 crypt32)
    target_compile_options(studio-assistant-tests PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(studio-assistant-tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME studio-assistant COMMAND studio-assistant-tests)
