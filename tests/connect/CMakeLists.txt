cmake_minimum_required(VERSION 3.16)
project(BambuConnectChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
enable_testing()
add_executable(bambu-connect-runtime-tests ../bambu_connect_runtime.test.cpp)
target_include_directories(bambu-connect-runtime-tests PRIVATE "${ROOT}/src" "${ROOT}/tests")
add_test(NAME connect-runtime COMMAND bambu-connect-runtime-tests)
set_tests_properties(connect-runtime PROPERTIES WORKING_DIRECTORY "${ROOT}")
if(WIN32)
    add_executable(bambu-connect-tests ../bambu_connect_windows.test.cpp ../slic3rutils/bambu_connect_tests.cpp)
    target_include_directories(bambu-connect-tests PRIVATE "${ROOT}/src" "${ROOT}/tests")
    target_compile_options(bambu-connect-tests PRIVATE /utf-8)
    target_link_libraries(bambu-connect-tests PRIVATE advapi32 ole32 user32)
    add_test(NAME connect-native COMMAND bambu-connect-tests)
    add_executable(bambu-connect-probe ../bambu_connect_probe.cpp "${ROOT}/src/slic3r/GUI/BambuConnectBridgeWin.cpp")
    target_include_directories(bambu-connect-probe PRIVATE "${ROOT}/src")
    target_compile_options(bambu-connect-probe PRIVATE /utf-8)
    target_link_libraries(bambu-connect-probe PRIVATE advapi32 ole32 user32)
endif()
